source = input(close)
responsiveness = math.max(0.00001, input.float(0.9, minval=0.0, maxval=1.0))
period = input(50)
sd = ta.stdev(source, 50) * responsiveness

var worm = source
diff = source - worm
delta = math.abs(diff) > sd ? math.sign(diff) * sd : diff
worm += delta
ma = ta.sma(source, period)
raw_momentum = (worm - ma) / worm
current_med = raw_momentum
min_med = ta.lowest(current_med, period)
max_med = ta.highest(current_med, period)
temp = (current_med - min_med) / (max_med - min_med)
value = 0.5 * 2
value *= (temp - .5 + .5 * nz(value[1]))
value := value > .9999 ? .9999 : value
value := value < -0.9999 ? -0.9999 : value
temp2 = (1 + value) / (1 - value)
momentum = .25 * math.log(temp2)
momentum += .5 * nz(momentum[1])

zigzag_func(prd) =>

    var int i = 1

    float ph = ta.highestbars(high, prd) == 0 ? high : na
    float pl = ta.lowestbars(low, prd) == 0 ? low : na
    var dir = 0
    iff_1 = pl and na(ph) ? -1 : dir
    dir := ph and na(pl) ? 1 : iff_1
    var max_array_size = 20
    var zigzag = array.new_float(0)
    oldzigzag = array.copy(zigzag)

    bool dirchanged = dir != dir[1]
    if ph or pl
        if dirchanged
            //add_to_zigzag(dir == 1 ? ph : pl, bar_index)
            value = dir == 1 ? ph : pl
            bindex = bar_index
            array.unshift(zigzag, bindex)
            array.unshift(zigzag, value)
            if array.size(zigzag) > max_array_size
                array.pop(zigzag)
                array.pop(zigzag)
        else
            //update_zigzag(dir == 1 ? ph : pl, bar_index)
            value = dir == 1 ? ph : pl
            bindex = bar_index
            if array.size(zigzag) == 0
                array.unshift(zigzag, bindex)
                array.unshift(zigzag, value)
                if array.size(zigzag) > max_array_size
                    array.pop(zigzag)
                    array.pop(zigzag)
            else
                if (dir == 1 and value > array.get(zigzag, 0)) or (dir == -1 and value < array.get(zigzag, 0))
                    array.set(zigzag, 0, value)
                    array.set(zigzag, 1, bindex)
                0.
    
    [zigzag, dir]

wedgec_bull_func(zigzag, dir)=>
    if array.size(zigzag) > 15
        // bull
        zigzag_ = array.copy(zigzag)
        if dir == 1
            array.remove(zigzag_, 0)
            array.remove(zigzag_, 0)

        A = array.get(zigzag_,6)

        C_index = array.get(zigzag_,3)
        D_index = array.get(zigzag_,1)

            line.delete(lineMap.get("bull_wedgec1"))
            line.delete(lineMap.get("bull_wedgec2"))
            line.delete(lineMap.get("bull_wedgec3"))

            label.delete(labelMap.get("bull_wedgec"))

        if X < A and C <= A and B >= D and B < A and X < B and D > X and close < (A - X)*0.382 + D//  and close[bar_index-int(D_index)] > ema200//and A - (A - X)*0.786 < B

            slope_high = (A - C)/(A_index - C_index)
            bh = C - slope_high*C_index
            bar_index_high = slope_high*int(bar_index) + bh
            x_index_high = slope_high*X_index + bh

            slope_low = (B - D)/(B_index - D_index)
            bl = B - slope_low*B_index
            bar_index_low = slope_low*int(bar_index) + bl
            x_index_low = slope_low*A_index + bl

            mx = math.max(slope_high, slope_low)
            mn = math.min(slope_high, slope_low)

            max = A
            min = D

            range_ = (max - min)/100
            a_norm = (A - min) / range_
            c_norm = (C - min) / range_
            b_norm = (B - min) / range_
            d_norm = (D - min) / range_

            slope1 = (a_norm - c_norm)/(A_index - C_index)
            slope2 = (b_norm - d_norm)/(B_index - D_index)

            tan_alpha = (slope1 - slope2)/(1 + slope1 *slope2)
            alpha = math.atan(tan_alpha)

            pred_cond = 1

            if show_pred == false
                pred_cond := close > bar_index_high ? 1 : 0

            if alpha < -0.099 and pred_cond and slope_high*int(bar_index) + bh > slope_low*int(bar_index) + bl

                myMap.put("bull_wedgec", D)
                
                line.delete(lineMap.get("bull_wedgec1"))
                line.delete(lineMap.get("bull_wedgec2"))
                line.delete(lineMap.get("bull_wedgec3"))

                lineMap.put("bull_wedgec3", line.new(int(A_index), x_index_low, bar_index, bar_index_low, xloc = xloc.bar_index, color = color.green,style = line.style_solid, width = line_w) )

bull_wedger_func(zigzag, dir)=>

    if array.size(zigzag) > 15 and close > ema20
        // bull
        zigzag_ = array.copy(zigzag)
        if dir == 1
            array.remove(zigzag_, 0)
            array.remove(zigzag_, 0)

        X = array.get(zigzag_,10)
        A = array.get(zigzag_,8)
        B = array.get(zigzag_,6)
        C = array.get(zigzag_,4)
        D = array.get(zigzag_,2)
        E = array.get(zigzag_,0)

        X_index = array.get(zigzag_,11)
        A_index = array.get(zigzag_,9)
        B_index = array.get(zigzag_,7)
        C_index = array.get(zigzag_,5)
        D_index = array.get(zigzag_,3)
        E_index = array.get(zigzag_,1)

        if close < myMap.get("bull_wedger")

            line.delete(lineMap.get("bull_wedger1"))
            line.delete(lineMap.get("bull_wedger2"))

            label.delete(labelMap.get("bull_wedger"))

        if E <= C and C < A and D < B// and E_index - A_index < 150

            slope_high = (B - D)/(int(B_index) - int(D_index))
            bh = B - slope_high*int(B_index)
            bar_index_high = slope_high*bar_index + bh
            x_index_high = slope_high*A_index + bh

            slope_low = (A - E)/(A_index - E_index)
            bl = A - slope_low*A_index
            bar_index_low = slope_low*bar_index + bl
            x_index_low = slope_low*A_index + bl

            max = B
            min = C

            range_ = (max - min)/100
            a_norm = (A - min) / range_
            c_norm = (C - min) / range_
            b_norm = (B - min) / range_
            d_norm = (D - min) / range_
            e_norm = (E - min) / range_

            slope1 = (a_norm - c_norm)/(A_index - C_index)
            slope2 = (c_norm - e_norm)/(C_index - E_index)
            slope3 = (a_norm - e_norm)/(A_index - E_index)
            slope4 = (b_norm - d_norm)/(B_index - D_index)

            s1 = math.abs(slope1)
            s2 = math.abs(slope2)

            slope_ratio = math.max(s1, s2)/math.min(s1, s2)
            slope_diff = math.max(s1, s2) - math.min(s1, s2)

            theta = math.atan((slope3 - slope4) / (1 + slope3 * slope4))

            pred_cond = 1

            if show_pred == false
                pred_cond := close > bar_index_high ? 1 : 0

            if bar_index_low < bar_index_high and slope_diff < 0.3 and theta > 0.06 and pred_cond// and math.min(close[bar_index - int(C_index)], open[bar_index - int(C_index)]) >= slope_low*(C_index) + bl

                myMap.put("bull_wedger", E)
                
                line.delete(lineMap.get("bull_wedger1"))
                line.delete(lineMap.get("bull_wedger2"))

                label.delete(labelMap.get("bull_wedger"))
                
                lineMap.put("bull_wedger1", line.new(int(A_index), A, int(bar_index + 10), slope_low*(bar_index + 10) + bl, xloc = xloc.bar_index, color = color.green,style = line.style_solid, width = line_w) )
                lineMap.put("bull_wedger2", line.new(int(B_index), B, int(bar_index + 10), slope_high*(bar_index + 10) + bh, xloc = xloc.bar_index, color = color.green,style = line.style_solid, width = line_w) )
                
                
                labelMap.put("bull_wedger", label.new(int(E_index), low[bar_index-int(A_index)], text="wedge reversal", yloc = yloc.belowbar, style = label.style_label_up, color = color.green, textcolor = color.white))
