//@version=5
indicator("Matriks DeMark Pivot + EMA + OBV", overlay=true, max_bars_back=500)

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// INPUTS
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

pivotTf = input.timeframe("D", "Pivot Periyodu")

useClassicPrevOpen = input.bool(false, "Klasik DeMark: Önceki gün Open kullan", tooltip="Kapalı olursa Matriks kodundaki gibi önceki gün Close ile bugünkü Open karşılaştırılır.")

showPivots = input.bool(true, "Pivotları Göster")
showEma    = input.bool(true, "EMA'ları Göster")
showObv    = input.bool(false, "OBV'yi fiyat grafiğinde göster")

obvDivider = input.float(250000000.0, "OBV Bölücü")

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// DAILY DATA
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

dHighPrev    = request.security(syminfo.tickerid, pivotTf, high[1], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)
dLowPrev     = request.security(syminfo.tickerid, pivotTf, low[1], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)
dClosePrev   = request.security(syminfo.tickerid, pivotTf, close[1], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)
dOpenCurrent = request.security(syminfo.tickerid, pivotTf, open, gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)
dOpenPrev    = request.security(syminfo.tickerid, pivotTf, open[1], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)

pivotOpen = useClassicPrevOpen ? dOpenPrev : dOpenCurrent

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// DEMARK PIVOT
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

x = dClosePrev < pivotOpen ? dLowPrev + dLowPrev + dHighPrev + dClosePrev : dClosePrev > pivotOpen ? dHighPrev + dHighPrev + dLowPrev + dClosePrev : dClosePrev + dClosePrev + dHighPrev + dLowPrev

pp = x / 4.0

rng = dHighPrev - dLowPrev

r1 = 2.0 * pp - dLowPrev
s1 = 2.0 * pp - dHighPrev

r2 = pp + rng
s2 = pp - rng

r3 = r1 + rng
s3 = s1 - rng

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// EMA
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ema21  = ta.ema(close, 21)
ema33  = ta.ema(close, 33)
ema55  = ta.ema(close, 55)
ema144 = ta.ema(close, 144)
ema233 = ta.ema(close, 233)
ema5   = ta.ema(close, 5)

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// OBV
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

obvRaw = ta.cum(close > close[1] ? volume : close < close[1] ? -volume : 0)
obv = obvRaw / obvDivider

obvHhv50 = ta.highest(obv, 50)
obvHhv25 = ta.highest(obv, 25)
obvHhv10 = ta.highest(obv, 10)

ro1 = obvHhv50[1]
ro2 = obvHhv25[1]
ro3 = obvHhv10[1]

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// PLOTS - PIVOT
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

plot(showPivots ? r3 : na, "R3", color=color.red, linewidth=1)
plot(showPivots ? r2 : na, "R2", color=color.new(color.red, 20), linewidth=1)
plot(showPivots ? r1 : na, "R1", color=color.new(color.red, 40), linewidth=1)

plot(showPivots ? pp : na, "PP", color=color.yellow, linewidth=2)

plot(showPivots ? s1 : na, "S1", color=color.new(color.lime, 40), linewidth=1)
plot(showPivots ? s2 : na, "S2", color=color.new(color.lime, 20), linewidth=1)
plot(showPivots ? s3 : na, "S3", color=color.lime, linewidth=1)

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// PLOTS - EMA
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

plot(showEma ? ema21 : na, "EMA 21", color=color.aqua, linewidth=1)
plot(showEma ? ema33 : na, "EMA 33", color=color.blue, linewidth=1)
plot(showEma ? ema55 : na, "EMA 55", color=color.purple, linewidth=1)
plot(showEma ? ema144 : na, "EMA 144", color=color.orange, linewidth=2)
plot(showEma ? ema233 : na, "EMA 233", color=color.maroon, linewidth=2)
plot(showEma ? ema5 : na, "EMA 5", color=color.white, linewidth=1)

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// PLOTS - OBV
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

plot(showObv ? obv : na, "OBV / Divider", color=color.gray, linewidth=1)
plot(showObv ? ro1 : na, "OBV HHV 50 Ref", color=color.fuchsia, linewidth=1)
plot(showObv ? ro2 : na, "OBV HHV 25 Ref", color=color.new(color.fuchsia, 30), linewidth=1)
plot(showObv ? ro3 : na, "OBV HHV 10 Ref", color=color.new(color.fuchsia, 60), linewidth=1)
